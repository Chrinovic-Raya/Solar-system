```python 
from picamera2 import Picamera2
import time

# Initialize the camera
picam2 = Picamera2()

# Configure camera resolution
config = picam2.create_preview_configuration(main={"size": (640, 480)})
picam2.configure(config)

# Start the camera capture stream
picam2.start()
print("Camera initialized. Capturing image in 2 seconds...")
time.sleep(2)

# Capture a still JPEG file
picam2.capture_file("opencv_test.jpg")

# Cleanly release camera resources
picam2.stop()
picam2.close()
print("Image saved successfully as opencv_test.jpg!")
```
